8// The positions are advanced by one full time step (dt), and displacement vectors
9// (dr) for one time step are updated. The displacement since the last neighbor
10// list creation (stored in p_nbrlist->dr) is also updated for each particle.
11void update_positions(struct Parameters *p_parameters, struct Nbrlist *p_nbrlist, struct Vectors *p_vectors)
12{

14 struct Vec3D *r = p_vectors->r; // Particle positions
15 struct Vec3D *dr = p_vectors->dr; // Displacement in one timestep
16 struct Vec3D *v = p_vectors->v; // Particle velocities

18 size_t num_part = p_parameters->num_part;

20
21 // Loop over all particles to update their positions
22 for (size_t i = 0; i < num_part; i++)

32 }

35// This function updates particle velocities by half a time step using the current forces.
36// The updated velocities are used in the velocity-Verlet integration scheme.
37// The function also calculates and returns the kinetic energy of the system.
38double update_velocities_half_dt(struct Parameters *p_parameters, struct Nbrlist *p_nbrlist, struct Vectors *p_vectors)
39{
40 double Ekin = 0.0; // Initialize kinetic energy
41 (void)p_nbrlist; // the neighbor list is not needed here
42 double dt_hlf = p_parameters->dt * 0.5; // Half time step
43 struct Vec3D *v = p_vectors->v; // Particle velocities
44 struct Vec3D *f = p_vectors->f; // Forces acting on particles
46 const double m = p_parameters->mass;
47 const double factor = dt_hlf / m; // Factor for velocity update
48
49 // Loop over all particles and update their velocities
50 for (size_t i = 0; i < p_parameters->num_part; i++)
51 {
52 v[i] = v3_add(v[i], v3_scl(factor, f[i]));
53 Ekin += (0.5*m)*v3_dot(v[i], v[i]); // Accumulate kinetic energy
54 }
55
56 // Kinetic (ideal-gas) contribution to the pressure: sum(m v^2)/(3V) = 2 Ekin/(3V)
57 double V = p_parameters->L.x * p_parameters->L.y * p_parameters->L.z;
58 p_vectors->press_kin = 2.0 * Ekin / (3.0 * V);
59
60 return Ekin; // Return the system's kinetic energy
61}

63// This function applies periodic boundary conditions to ensure particles stay inside the simulation box.
64// If a particle moves beyond the box, it is wrapped around to the opposite side.
65// Keeping all positions inside [0,L) is what allows the cheap minimum image
66// convention (v3_min_image) used everywhere else in the code.
67void boundary_conditions(struct Parameters *p_parameters, struct Vectors *p_vectors)
68{
69 struct Vec3D invL; // Inverse of the box size
70 struct Vec3D *r = p_vectors->r; // Particle positions
71 struct Vec3D L = p_parameters->L; // Box dimensions
72 size_t num_part = p_parameters->num_part; // Number of particles
73
74 invL.x = 1.0 / L.x;
75 invL.y = 1.0 / L.y;
76 invL.z = 1.0 / L.z;
77
78 // Loop over all particles and apply periodic boundary conditions
79 for (size_t i = 0; i < num_part; i++)
80 {
81 r[i].x -= L.x * floor(r[i].x * invL.x); // Apply periodic boundary in x-direction
82 r[i].y -= L.y * floor(r[i].y * invL.y); // Apply periodic boundary in y-direction
83 r[i].z -= L.z * floor(r[i].z * invL.z); // Apply periodic boundary in z-direction
84 }
85}
