71// Save the state of the simulation to a binary restart file: the number of
72// particles followed by the position, velocity and force arrays. Parameters,
73// types and topology are NOT stored; they are set up again when restarting
74// (see main.c), so the restart file stays valid when settings change.
75void save_restart(struct Parameters *p_parameters, struct Vectors *p_vectors)
76{
77 FILE* p_file = fopen( p_parameters->restart_out_filename, "wb");
78 size_t num_part = p_parameters->num_part;
79 size_t sz = num_part*sizeof(struct Vec3D);
80
81 fwrite(&num_part, sizeof(size_t), 1, p_file);
82 fwrite(p_vectors->r, sz, 1, p_file);
83 fwrite(p_vectors->v ,sz, 1, p_file);
84 fwrite(p_vectors->f, sz, 1, p_file);
85 fclose(p_file);
