111double virial_test(ForceKernel kernel,
112 struct Parameters *p_parameters,
113 struct Nbrlist *p_nbrlist,
114 struct Vectors *p_vectors)
115{
116 // Validate the accumulated virial against a numerical derivative of the
117 // potential energy under isotropic scaling of all positions and the box:
118 // r -> lambda r, L -> lambda L. For such scaling W = sum(f.r) = -dU/dlambda
119 // at lambda = 1, so 3V (press_vir_nb + press_vir_bnd) must match -dU/dlambda.
120 size_t num_part = p_parameters->num_part;
121 Vec3D *r = p_vectors->r;
122 Vec3D *r_saved = malloc(num_part * sizeof(Vec3D));
123 if (!r_saved)
124 {
125 fprintf(stderr, "virial_test: allocation failed\n");
126 return -1.0;
127 }
128 memcpy(r_saved, r, num_part * sizeof(Vec3D));
129 Vec3D L_saved = p_parameters->L;
130
131 const double h = 1e-6;
132 double U[2];
133 for (int sgn = 0; sgn < 2; ++sgn)
134 {
135 double lambda = (sgn == 0) ? 1.0 + h : 1.0 - h;
136 for (size_t i = 0; i < num_part; ++i)
137 r[i] = v3_scl(lambda, r_saved[i]);
138 p_parameters->L = v3_scl(lambda, L_saved);
139 build_nbrlist(p_parameters, p_vectors, p_nbrlist); // full rebuild: L changed
140 U[sgn] = kernel(p_parameters, p_nbrlist, p_vectors);
141 }
142
143 // Restore the original state and recompute forces and virials there
144 memcpy(r, r_saved, num_part * sizeof(Vec3D));
145 p_parameters->L = L_saved;
146 build_nbrlist(p_parameters, p_vectors, p_nbrlist);
147 kernel(p_parameters, p_nbrlist, p_vectors);
148 free(r_saved);
149
150 double W_fd = -(U[0] - U[1]) / (2.0 * h);
151 double V = L_saved.x * L_saved.y * L_saved.z;
152 double W_an = 3.0 * V * (p_vectors->press_vir_nb + p_vectors->press_vir_bnd);
153 // Note: angle and dihedral potentials depend only on directions, which are
154 // invariant under isotropic scaling, so their exact virial is zero. To avoid
155 // a meaningless relative error when W is (near) zero, the difference is
156 // normalized by the energy scale in that case.
157 double scale = fmax(fabs(W_fd), fabs(U[0]));
158 double result = fabs(W_an - W_fd) / ((scale > 0.0) ? scale : 1.0);
159
160 printf("Virial test: analytical W = %g, finite-difference W = %g, rel.err=%g\n",
161 W_an, W_fd, result);
162 return result;
163}
