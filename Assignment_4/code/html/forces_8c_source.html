9// Compute all forces and torques on the particles: gravity, particle-particle
10// contacts and particle-wall contacts.
11// This function returns the total potential energy of the system.
12double calculate_forces(struct Parameters *p_parameters, struct Nbrlist *p_nbrlist, struct Colllist *p_colllist, struct Vectors *p_vectors)
13{
14 double Epot = 0.0;
15 struct Vec3D *f = p_vectors->f;
16 struct Vec3D *T = p_vectors->T;
17 struct Vec3D g;
18 g.x = p_parameters->g.x;
19 g.y = p_parameters->g.y;
20 g.z = p_parameters->g.z;
21 // In a box that is oscillated vertically the calculation is done in the
22 // frame of the box, which adds the inertial acceleration of that frame to
23 // gravity. With the default amplitude A = 0 the box is at rest and this
24 // term vanishes. See cases/DEM_BNE for a vibrated bed.
25 if (p_parameters->A != 0.0)
26 {
27 double omega = 2.0 * PI * p_parameters->freq;
28 g.z -= p_parameters->A * omega * omega * sin(omega * p_vectors->time);
29 }
30 double * mass = p_vectors->mass;
31 // Initialize the force of each particle to gravity and its torque to zero;
32 // the contact routines below then add their contributions
33 const size_t num_part = p_parameters->num_part;
34 for (size_t i = 0; i < num_part; i++)
35 {
36 f[i].x = mass[i]*g.x;
37 f[i].y = mass[i]*g.y;
38 f[i].z = mass[i]*g.z;
39 T[i] = (struct Vec3D){0.0, 0.0, 0.0};
40 }
41 Epot += calculate_forces_pp(p_parameters, p_nbrlist, p_colllist, p_vectors);
42 Epot += calculate_forces_pw(p_parameters, p_colllist, p_vectors);
43 return Epot;
44}

46// Compute the forces and torques due to particle-particle contacts, walking
47// the touching pairs collected by update_colllist.
48//
49// The contact model is the linear soft-sphere model: two overlapping spheres
50// feel, in the normal direction, a spring force proportional to their overlap
51// plus a dashpot force proportional to their approach velocity (which is what
52// makes a collision dissipative and gives a restitution coefficient below
53// one). In the tangential direction a spring acting on the accumulated
54// tangential displacement of the contact models sticking friction, again with
55// a dashpot; when that force exceeds the Coulomb limit fric*|Fn|, the contact
56// slides and the tangential force is capped at the limit.
57//
58// This function returns the potential energy of (the conservative part of)
59// these interactions: the energy in the normal and tangential springs.
60double calculate_forces_pp(struct Parameters *p_parameters, struct Nbrlist *p_nbrlist, struct Colllist *p_colllist, struct Vectors *p_vectors)
61{
62 double Epot = 0.0;
63 const double k_n_pp = p_parameters->k_n_pp;
64 const double eta_n_pp = p_parameters->eta_n_pp;
65 const double k_t_pp = p_parameters->k_t_pp;
66 const double eta_t_pp = p_parameters->eta_t_pp;
67 const double fric_pp = p_parameters->fric_pp;
68 struct Pair *nbr = p_nbrlist->nbr;
69 struct DeltaR *tijs = p_nbrlist->tij;
70 const size_t num_contacts = p_colllist->num_contacts;
71 const size_t *contact = p_colllist->contact;
72 double * R = p_vectors->radius;
73 struct Vec3D *f = p_vectors->f;
74 struct Vec3D *T = p_vectors->T;
75 struct Vec3D *v = p_vectors->v;
76 struct Vec3D *omega = p_vectors->omega;
77 double * mass = p_vectors->mass;
78 double inv_mass_ref = 1.0/p_parameters->mass_ref;
79 for (size_t c = 0; c < num_contacts; c++)
80 {
81 // for each touching pair compute the contact forces
82 size_t k = contact[c];
83 struct DeltaR rij = nbr[k].rij;
84 size_t i = nbr[k].i;
85 size_t j = nbr[k].j;
86 double sumR = R[i] + R[j];
87 // The dashpot constants were tuned for a pair of reference mass; scaling
88 // them with sqrt(m_eff/m_ref), with m_eff the reduced mass of the pair,
89 // keeps the restitution coefficient the same for all particle sizes.
90 double mass_factor = sqrt(2.0*mass[i]*mass[j]/(mass[i]+mass[j])*inv_mass_ref);
91 // normal spring force, proportional to the overlap of the two spheres
92 double r = sqrt(rij.sq);
93 double overlap = sumR - r;
94 double fr = k_n_pp * overlap / r;
95 struct DeltaR dfn;
96 dfn.x = fr * rij.x;
97 dfn.y = fr * rij.y;
98 dfn.z = fr * rij.z;
99 Epot += 0.5 * k_n_pp * overlap * overlap;
100 // normal dashpot force
101 struct Vec3D vij, vijn;
102 vij.x = v[i].x - v[j].x;
103 vij.y = v[i].y - v[j].y;
104 vij.z = v[i].z - v[j].z;
105 double fctr = (vij.x * rij.x + vij.y * rij.y + vij.z * rij.z) / rij.sq;
106 vijn.x = fctr * rij.x;
107 vijn.y = fctr * rij.y;
108 vijn.z = fctr * rij.z;
109 fr = -mass_factor*eta_n_pp;
110 dfn.x += fr * vijn.x;
111 dfn.y += fr * vijn.y;
112 dfn.z += fr * vijn.z;
113 // tangential spring force
114 struct DeltaR tij = tijs[k];
115 fr = -k_t_pp;
116 struct DeltaR dft;
117 dft.x = fr * tij.x;
118 dft.y = fr * tij.y;
119 dft.z = fr * tij.z;
120 // tangential dashpot force, acting on the relative velocity of the two
121 // *surfaces* at the contact point: the tangential part of the velocity
122 // difference plus the contribution of both rotations
123 struct Vec3D vijt;
124 vijt.x = vij.x - vijn.x;
125 vijt.y = vij.y - vijn.y;
126 vijt.z = vij.z - vijn.z;
127 vijt.x -= 0.5 * ((omega[i].y + omega[j].y) * rij.z -
128 (omega[i].z + omega[j].z) * rij.y);
129 vijt.y -= 0.5 * ((omega[i].z + omega[j].z) * rij.x -
130 (omega[i].x + omega[j].x) * rij.z);
131 vijt.z -= 0.5 * ((omega[i].x + omega[j].x) * rij.y -
132 (omega[i].y + omega[j].y) * rij.x);
133 fr = -mass_factor*eta_t_pp;
134 dft.x += fr * vijt.x;
135 dft.y += fr * vijt.y;
136 dft.z += fr * vijt.z;
137
138 //If tangential force is too large then sliding takes place
139 dfn.sq = dfn.x * dfn.x + dfn.y * dfn.y + dfn.z * dfn.z;
140 dft.sq = dft.x * dft.x + dft.y * dft.y + dft.z * dft.z;
141 if (dft.sq >= fric_pp * fric_pp * dfn.sq) //sliding
142 {
143 fr = fric_pp * sqrt(dfn.sq / dft.sq);
144 dft.x *= fr;
145 dft.y *= fr;
146 dft.z *= fr;
147 // when sliding set the tangential displacement such that the sticking force (nearly) equals the sliding force
148 tijs[k].x = -dft.x / k_t_pp;
149 tijs[k].y = -dft.y / k_t_pp;
150 tijs[k].z = -dft.z / k_t_pp;
151 }
152 else
153 Epot += 0.5 * k_t_pp * tij.sq;
154 struct Vec3D df;
155 df.x = dfn.x + dft.x;
156 df.y = dfn.y + dft.y;
157 df.z = dfn.z + dft.z;
158 // Torque = lever arm x tangential force, with the contact point taken
159 // halfway along rij. Both particles get the same torque: the tangential
160 // force on j is opposite, but so is its lever arm.
161 struct Vec3D dT;
162 dT.x = 0.5 * (rij.z * dft.y - rij.y * dft.z);
163 dT.y = 0.5 * (rij.x * dft.z - rij.z * dft.x);
164 dT.z = 0.5 * (rij.y * dft.x - rij.x * dft.y);
165 f[i].x += df.x;
166 f[i].y += df.y;
167 f[i].z += df.z;
168 f[j].x -= df.x;
169 f[j].y -= df.y;
170 f[j].z -= df.z;
171 T[i].x += dT.x;
172 T[i].y += dT.y;
173 T[i].z += dT.z;
174 T[j].x += dT.x;
175 T[j].y += dT.y;
176 T[j].z += dT.z;
177 }
178 p_vectors->num_contacts = num_contacts;
179 return Epot;
180}

182// Compute the forces and torques due to particle-wall contacts, walking the
183// contacts collected by update_colllist. The contact model is the same
184// linear soft-sphere model as in calculate_forces_pp, with the wall taking the
185// role of an infinitely heavy, flat particle; each wall has its own contact
186// parameters, indexed by wall id.
187// This function returns the potential energy of (the conservative part of)
188// these interactions.
189double calculate_forces_pw(struct Parameters *p_parameters, struct Colllist *p_colllist, struct Vectors *p_vectors)
190{
191 double Epot = 0.0;
192 double *k_n_pw = p_parameters->k_n_pw;
193 double *eta_n_pw = p_parameters->eta_n_pw;
194 double *k_t_pw = p_parameters->k_t_pw;
195 double *eta_t_pw = p_parameters->eta_t_pw;
196 double *fric_pw = p_parameters->fric_pw;
197 size_t num_w = p_colllist->num_w;
198 size_t *indcs_w = p_colllist->indcs_w;
199 unsigned int *wall_id = p_colllist->wall_id;
200 struct DeltaR *riw = p_colllist->riw;
201 struct DeltaR *tiw = p_colllist->tiw;
202 struct Vec3D *vw = p_colllist->vw;
203 double * R = p_vectors->radius;
204 struct Vec3D *f = p_vectors->f;
205 struct Vec3D *T = p_vectors->T;
206 struct Vec3D *v = p_vectors->v;
207 struct Vec3D *omega = p_vectors->omega;
208 double * mass = p_vectors->mass;
209 double inv_mass_ref = 1.0/p_parameters->mass_ref;
210 for (size_t j = 0; j < num_w; ++j)
211 {
212 unsigned int w_id = wall_id[j];
213 size_t i = indcs_w[j];
214 struct DeltaR rij = riw[j];
215 //normal elastic force
216 double r = sqrt(rij.sq);
217 double overlap = R[i] - r;
218 double mass_factor = sqrt(mass[i]*inv_mass_ref);
219 double fr = k_n_pw[w_id] * overlap / r;
220 struct DeltaR dfn;
221 dfn.x = fr * rij.x;
222 dfn.y = fr * rij.y;
223 dfn.z = fr * rij.z;
224 Epot += 0.5 * k_n_pw[w_id] * overlap * overlap;
225 // normal dashpot force
226 struct Vec3D vij;
227 vij.x = v[i].x - vw[j].x;
228 vij.y = v[i].y - vw[j].y;
229 vij.z = v[i].z - vw[j].z;
230 double fctr = (vij.x * rij.x + vij.y * rij.y + vij.z * rij.z) / rij.sq;
231 struct Vec3D vijn;
232 vijn.x = fctr * rij.x;
233 vijn.y = fctr * rij.y;
234 vijn.z = fctr * rij.z;
235 fr = -mass_factor*eta_n_pw[w_id];
236 dfn.x += fr * vijn.x;
237 dfn.y += fr * vijn.y;
238 dfn.z += fr * vijn.z;
239
240 struct DeltaR tij = tiw[i * NUM_WALLS_MAX + w_id];
241 fr = -k_t_pw[w_id];
242 struct DeltaR dft;
243 dft.x = fr * tij.x;
244 dft.y = fr * tij.y;
245 dft.z = fr * tij.z;
246 struct Vec3D vijt;
247 vijt.x = vij.x - vijn.x;
248 vijt.y = vij.y - vijn.y;
249 vijt.z = vij.z - vijn.z;
250 vijt.x -= (omega[i].y * rij.z - omega[i].z * rij.y);
251 vijt.y -= (omega[i].z * rij.x - omega[i].x * rij.z);
252 vijt.z -= (omega[i].x * rij.y - omega[i].y * rij.x);
253 fr = -mass_factor*eta_t_pw[w_id];
254 dft.x += fr * vijt.x;
255 dft.y += fr * vijt.y;
256 dft.z += fr * vijt.z;
257
258 dfn.sq = dfn.x * dfn.x + dfn.y * dfn.y + dfn.z * dfn.z;
259 dft.sq = dft.x * dft.x + dft.y * dft.y + dft.z * dft.z;
260 if (dft.sq >= fric_pw[w_id] * fric_pw[w_id] * dfn.sq)
261 {
262 //If tangential force is too large then sliding takes place:
263 double fr = fric_pw[w_id] * sqrt(dfn.sq / dft.sq);
264 dft.x *= fr;
265 dft.y *= fr;
266 dft.z *= fr;
267 // when sliding set the tangential displacement such that the sticking force (nearly) equals the sliding force
268 struct DeltaR *t = &tiw[i * NUM_WALLS_MAX + w_id];
269 t->x = -dft.x / k_t_pw[w_id];
270 t->y = -dft.y / k_t_pw[w_id];
271 t->z = -dft.z / k_t_pw[w_id];
272 t->sq = t->x * t->x + t->y * t->y + t->z * t->z;
273 }
274 else
275 Epot += 0.5 * k_t_pw[w_id] * tij.sq;
276 f[i].x += dfn.x + dft.x;
277 f[i].y += dfn.y + dft.y;
278 f[i].z += dfn.z + dft.z;
279 T[i].x += (rij.z * dft.y - rij.y * dft.z);
280 T[i].y += (rij.x * dft.z - rij.z * dft.x);
281 T[i].z += (rij.y * dft.x - rij.x * dft.y);
282 }
283 return Epot;
