156bool silo_bottom_wall(struct Parameters *p_parameters, double radius, struct Vec3D *r, struct DeltaR *riw, struct Vec3D *vw)
157{
158 // While the orifice is closed (orifice_open = 0) the plate is a full
159 // plane, which is the given part below: it works like bottom_wall, at
160 // height z_silo. When the orifice is open, a particle whose centre is over
161 // the hole falls through, unless it touches the edge of the hole.
162 double dz = r->z - p_parameters->z_silo; // keep the sign: under the plate riw points down
163 if (fabs(dz) >= radius)
164 return false; // any contact, flat or edge, needs the plane within one radius
165
166 double dx = r->x - 0.5 * p_parameters->L.x; // horizontal offset from the axis
167 double dy = r->y - 0.5 * p_parameters->L.y;
168 double rc = sqrt(dx * dx + dy * dy); // distance of the centre to the axis
169
170 if (!p_parameters->orifice_open || rc >= p_parameters->R_orifice)
171 {
172 // Contact with the flat part of the plate (or the closed plate).
173 *riw = (struct DeltaR){0.0, 0.0, dz, dz * dz};
174 *vw = (struct Vec3D){0};
175 return true;
176 }
177
188 return false;
189}
