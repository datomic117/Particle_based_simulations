76int main(void)
77{
78 struct Vectors vectors;
79 struct Parameters parameters;
80 struct Nbrlist nbrlist;
81 struct Colllist colllist;
82 size_t step = 0;
83 double Ekin, Epot;
84
85 set_parameters(&parameters);
86 alloc_memory(&parameters, &vectors, &nbrlist, &colllist);
87 if(parameters.load_restart == 1)
88 {
89 load_restart(&parameters, &vectors);
90 }
91 else
92 initialise(&parameters, &vectors);
93
94 // Put every particle inside the box before the cell list is built for the
95 // first time. build_celllist turns a position straight into a cell index, so
96 // a position outside the box would index outside the array of cells. During
97 // the loop below boundary_conditions keeps this true; here it has to be
98 // established, because an initial configuration or a restart file is free to
99 // put a particle anywhere.
100 boundary_conditions(&parameters, &vectors);
101
102
103 build_nbrlist(&parameters, &vectors, &nbrlist);
104 update_colllist(&parameters, &vectors, &nbrlist, &colllist);
105 Epot = calculate_forces(&parameters, &nbrlist, &colllist, &vectors);
106 record_trajectories_xyz(1, &parameters, &vectors);
107
108 // Record the run and its geometry, so that the output can be interpreted on its own.
109 printf("# run %d box %g %g %g R_cyl %g z_silo %g R_orifice %g t_open %g num_part %zu dt %g\n",
110 parameters.run, parameters.L.x, parameters.L.y, parameters.L.z, parameters.R_cyl,
111 parameters.z_silo, parameters.R_orifice, parameters.t_open, parameters.num_part, parameters.dt);
112 printf("#%7s %15s %15s %15s %15s %15s %8s\n", "Step", "Time", "Z", "Epot", "Ekin", "Etot", "N_silo");
113
114 while (step < parameters.num_dt_steps) //start of the velocity-Verlet loop
115 {
116
117 step++;
118 vectors.time += parameters.dt;
119 update_parameters(&parameters, vectors.time);
120
121 Ekin = update_velocities_half_dt(&parameters, &nbrlist, &vectors);
122 update_positions(&parameters, &nbrlist, &vectors);
123
124 update_tangential_displacements(&parameters, &vectors, &nbrlist, &colllist);
125 boundary_conditions(&parameters, &vectors);
126 update_nbrlist(&parameters, &vectors, &nbrlist);
127 update_colllist(&parameters, &vectors, &nbrlist, &colllist);
128 Epot = calculate_forces(&parameters, &nbrlist, &colllist, &vectors);
129 Ekin = update_velocities_half_dt(&parameters, &nbrlist, &vectors);
130
131 // Print to the screen to monitor the progress of the simulation. Z is the
132 // coordination number, the average number of contacts per particle.
133 if (step%parameters.num_dt_printf ==0)
134 printf("%8lu %15.7g %15.7g %15.7g %15.7g %15.7g %8zu\n", (long unsigned) step, vectors.time,
135 2.0*((double) vectors.num_contacts)/((double) parameters.num_part),
136 Epot, Ekin, Epot+Ekin, count_in_silo(&parameters, &vectors));
137 if (step%parameters.num_dt_traj ==0) record_trajectories_xyz(0,&parameters,&vectors);
138 if (step%parameters.num_dt_restart == 0) save_restart(&parameters,&vectors);
139 }
140 save_restart(&parameters,&vectors);
141 free_memory(&vectors, &nbrlist, &colllist);
142
143 return 0;
144}
