68// Initializes particle positions on a cubic lattice based on the box dimensions and particle count.
69// Positions are evenly spaced, and the lattice size is adjusted to accommodate the largest particle.
70// This is the bed of Part B, and the stand-in for the parts not yet written.
71void initialise_positions(struct Parameters *p_parameters, struct Vectors *p_vectors)
72{
73 struct Vec3D dr;
74 struct Index3D n;
75 double dl;
76 size_t ipart;
77 size_t num_part = p_parameters->num_part;
78 double *R = p_vectors->radius;
79 double R_max = 0;
80 for (ipart =0; ipart<num_part; ++ipart)
81 R_max = (R[ipart]>R_max? R[ipart]: R_max);
82 dl = 2.1*R_max; // lattice spacing just over one diameter of the largest particle, so no pair starts overlapping
83 n.i = (int)floor(p_parameters->L.x / dl);
84 n.j = (int)floor(p_parameters->L.y / dl);
85 n.k = (int) (p_parameters->num_part/(n.i*n.j)+1);
86 dr.x = p_parameters->L.x / (double)n.i;
87 dr.y = p_parameters->L.y / (double)n.j;
88 dr.z = dl;
89 ipart = 0;
90 for (size_t i = 0; i < n.i; ++i)
91 for (size_t j = 0; j < n.j; ++j)
92 for (size_t k = 0; k < n.k; ++k, ++ipart)
93 {
94 if (ipart >= num_part)
95 break;
96 p_vectors->r[ipart].x = (i + 0.5) * dr.x;
97 p_vectors->r[ipart].y = (j + 0.5) * dr.y;
98 p_vectors->r[ipart].z = (k + 0.5) * dr.z;
99 }
100}
