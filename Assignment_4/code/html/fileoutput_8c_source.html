8void record_trajectories_xyz(int reset, struct Parameters *p_parameters, struct Vectors *p_vectors)
9/* Write the particle positions to an xyz file, a text format that
10 visualization programs such as OVITO read directly. Besides the position,
11 each line carries the particle radius and speed as extra columns. The
12 second line of every frame is in the extended-XYZ convention: it names the
13 columns (Properties=...) and gives the box (Lattice=...) and the time, so
14 that OVITO recognises the radius and the box without a column mapping and
15 offers the speed as a property to colour by. All values are in SI units
16 (m, m/s, s), written with enough digits that a radius keeps five
17 significant figures.
18 The filename (without extension) is given by p_parameters->filename_xyz.
19 If reset = 1 the data is written to the file deleting data it possibly contained.
20 If reset = 0 the data is appended. */
21{
22 FILE *fp_traj;
23 // room for the configured stem plus the extension and the terminator
24 char filename[sizeof p_parameters->filename_xyz + 8];
25
26 snprintf(filename, sizeof filename, "%s%s", p_parameters->filename_xyz, ".xyz");
27 if (reset == 1)
28 {
29 fp_traj = fopen(filename, "w");
30 }
31 else
32 {
33 fp_traj = fopen(filename, "a");
34 }
35
36 struct Vec3D L = p_parameters->L;
37 fprintf(fp_traj, "%zu\n", p_parameters->num_part);
38 fprintf(fp_traj, "Lattice=\"%.8g 0 0 0 %.8g 0 0 0 %.8g\" "
39 "Properties=species:S:1:pos:R:3:radius:R:1:speed:R:1 Time=%.8g\n",
40 L.x, L.y, L.z, p_vectors->time);
41 struct Vec3D *r = p_vectors->r;
42 struct Vec3D *v = p_vectors->v;
43 double * R = p_vectors->radius;
44 for (size_t i = 0; i < p_parameters->num_part; i++)
45 {
46 fprintf(fp_traj, "C %.8f %.8f %.8f %.8f %.6f\n", r[i].x, r[i].y, r[i].z,
47 R[i], sqrt(v[i].x*v[i].x + v[i].y*v[i].y +v[i].z*v[i].z));
48 }
49
50 fclose(fp_traj);
51}

53void save_restart(struct Parameters *p_parameters, struct Vectors *p_vectors)
54/* Save the state of the simulation to a binary restart file: the time, the
55 number of particles, and the radius, mass, position, velocity, angular
56 velocity, force and torque arrays. Parameters are NOT stored; they are set
57 up again when restarting. Note that the tangential contact histories are not
58 stored either, so a restarted run is not a bit-for-bit continuation. */
59{
60 FILE* p_file = fopen( p_parameters->restart_out_filename, "wb");
61 size_t num_part = p_parameters->num_part;
62 size_t sz = num_part*sizeof(struct Vec3D);
63 fwrite(&p_vectors->time, sizeof(double), 1, p_file);
64 fwrite(&num_part, sizeof(size_t), 1, p_file);
65 fwrite(p_vectors->radius, num_part*sizeof(double), 1, p_file);
66 fwrite(p_vectors->mass, num_part*sizeof(double), 1, p_file);
67 fwrite(p_vectors->r, sz, 1, p_file);
68 fwrite(p_vectors->v ,sz, 1, p_file);
69 fwrite(p_vectors->omega, sz, 1, p_file);
70 fwrite(p_vectors->f, sz, 1, p_file);
71 fwrite(p_vectors->T, sz, 1, p_file);
72 fclose(p_file);
73}
