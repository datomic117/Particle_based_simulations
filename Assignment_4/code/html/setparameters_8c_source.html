11 * Choosing a run. Every setting below has a default, and the ones you change
12 * from run to run can also be given on the command line, so that every run is
13 * one reproducible compile command, for example
14 *
15 * gcc -O3 -DRUN=1 -DNUM_TEST=2 -DV_IMPACT=0.3 *.c -o dem -lm Part A, head-on pair
16 * gcc -O3 -DRUN=2 *.c -o dem -lm Part B, tilted bed
17 * gcc -O3 -DRUN=3 -DD_ORIFICE=8 *.c -o dem -lm Part D, orifice of 8 d_p
18 *
19 * RUN 1, 2 and 3 are RUN_COLLISION, RUN_SLOPE and RUN_SILO of constants.h.

43void set_silo_geometry(struct Parameters *p_parameters)
44{
45 double aspect = 3.0; // H/R_cyl of the settled fill, H of about 1.5 silo diameters
46 double pack_frac = 0.6; // solids volume fraction of a settled random packing
47 double R_mean = 0.5 * (p_parameters->R_min + p_parameters->R_max);
48 double V_part = p_parameters->num_part * (4.0/3.0) * PI * R_mean * R_mean * R_mean;
49 p_parameters->R_cyl = pow(V_part / (pack_frac * PI * aspect), 1.0/3.0);
50 double H_fill = aspect * p_parameters->R_cyl;
51 p_parameters->z_silo = H_fill; // catch region below the plate, one fill height deep
52 p_parameters->R_orifice = 0.5 * D_ORIFICE * 2.0 * R_mean;
53 p_parameters->orifice_open = 0; // closed while the silo is filled and settles
54 // Above the plate the box holds the loose starting lattice: clipped to the
55 // cylinder it stacks to about 1.9 H_fill, and more
56 // for a small silo, so 2.5 fill heights leave headroom.
57 p_parameters->L = (struct Vec3D){4.0*p_parameters->R_cyl, 4.0*p_parameters->R_cyl,
58 p_parameters->z_silo + 2.5*H_fill};
59}

61void set_parameters(struct Parameters *p_parameters)
62/* Set the parameters of this simulation */
63{
64 p_parameters->run = RUN;
65 p_parameters->density = 2500; // mass density of the particle material, kg/m^3
66
67 double R_min = 0.975e-3;
68 double R_max = 1.05*R_min;
69 p_parameters->R_min = R_min; // minimum particle radius
70 p_parameters->R_max = R_max; // maximum particle radius; the mean diameter is 2.0 mm
71 double kn = 1000; // spring stiffness of normal contact force
72 double e_n_pp = 0.92, e_t_pp = 0.33, muf = 0.50; // restitution and friction coefficients for particle-particle contacts
73 double e_n_pw = 0.85, e_t_pw = 0.33, muf_w = 0.45; // the same for particle-wall contacts, equal for all walls
74
75 // The box of the starting code: periodic in x and y, a floor and a ceiling.
76 // Parts A and B use it; the silo sizes its own box in set_silo_geometry.
77 p_parameters->L = (struct Vec3D){4e1*R_min, 4e1*R_min, 1e2*R_max};
78 p_parameters->num_walls = 2;
79 p_parameters->wall_function[0] = bottom_wall;
80 p_parameters->wall_function[1] = top_wall;
81 p_parameters->g = (struct Vec3D){0.0, 0.0, -GRAVITY};
82
83 // Fields of the other set-ups, given a value so that nothing is left unset.
84 p_parameters->v_impact = 0.0;
85 p_parameters->t_tilt = 0.0;
86 p_parameters->tilt_rate = 0.0;
87 p_parameters->slope_angle = 0.0;
88 p_parameters->R_cyl = 0.0;
89 p_parameters->z_silo = 0.0;
90 p_parameters->R_orifice = 0.0;
91 p_parameters->orifice_open = 0;
92 p_parameters->t_open = 0.0;
93
94 double t_end; // simulated time of the run; num_dt_steps follows once dt is known
95 switch (p_parameters->run)
96 {
97 case RUN_COLLISION:
98 // Part A: one or two particles, no gravity, so that the kinetic energy
99 // before and after the collision gives the restitution coefficient.
100 p_parameters->num_part = NUM_TEST;
101 if (NUM_TEST != 1 && NUM_TEST != 2)
102 {
103 fprintf(stderr, "set_parameters: NUM_TEST must be 1 (wall) or 2 (pair), not %d\n", NUM_TEST);
104 exit(EXIT_FAILURE);
105 }
106 p_parameters->v_impact = V_IMPACT;
107 p_parameters->g = (struct Vec3D){0.0, 0.0, 0.0};
108 t_end = 2e-3; // about ten contact times
109 p_parameters->num_dt_printf = 1; // every step, to resolve the collision
110 p_parameters->num_dt_traj = 1000000000; // no trajectory needed
111 break;
112 case RUN_SLOPE:
113 // Part B: a bed several layers thick settles under vertical gravity until
114 // t_tilt, after which gravity rotates at a constant rate (update_parameters).
115 p_parameters->num_part = 2000;
116 p_parameters->t_tilt = 0.3;
117 p_parameters->tilt_rate = TILT_RATE * PI / 180.0;
118 t_end = p_parameters->t_tilt + 24.0 / TILT_RATE; // ramp from 0 to 24 degrees
119 p_parameters->num_dt_printf = 100;
120 p_parameters->num_dt_traj = 2000; // a frame every 0.022 s
121 break;
122 case RUN_SILO:
123 // Parts C and D: the fill settles on the closed plate until t_open, then
124 // the orifice opens (update_parameters) and the silo discharges.
125 p_parameters->num_part = 7500;
126 set_silo_geometry(p_parameters);
127 p_parameters->num_walls = 4;
128 p_parameters->wall_function[2] = cylinder_wall; // the silo wall (Task C1)
129 p_parameters->wall_function[3] = silo_bottom_wall; // the silo bottom plate with the orifice (Task C2)
130 p_parameters->t_open = 0.3;
131 t_end = p_parameters->t_open + T_DISCHARGE;
132 p_parameters->num_dt_printf = 200; // a line every 2.2 ms
133 p_parameters->num_dt_traj = 2000; // a frame every 0.022 s
134 break;
135 default:
136 fprintf(stderr, "set_parameters: unknown run %d\n", p_parameters->run);
137 exit(EXIT_FAILURE);
138 }
139
140 // The box is at rest; set A and freq to oscillate it vertically instead.
141 p_parameters->A = 0.0;
142 p_parameters->freq = 0.0;
143
144 strcpy(p_parameters->filename_xyz, "trajectories"); //filename (without extension) for the xyz file
145 p_parameters->load_restart = 0; //if equal 1 restart file is loaded
146 strcpy(p_parameters->restart_in_filename, "restart.dat"); //filename for loaded restart file
147 p_parameters->num_dt_restart = 1000000000; // number of time steps between saves; the end state is always saved
148 strcpy(p_parameters->restart_out_filename, "restart.dat"); //filename for saved restart file

150 // During a contact the normal overlap behaves as a damped harmonic
151 // oscillator, which can be solved analytically: half an oscillation period is
152 // the contact time, and the damping decides how much of the approach velocity
153 // remains at rebound, i.e. the restitution coefficient. The expressions below
154 // run that solution backwards: from the spring constant kn and the prescribed
155 // restitution coefficients they compute the contact time and the dashpot and
156 // tangential-spring constants that reproduce those coefficients, for a
157 // contact of two particles of the reference mass. (In the force routines the
158 // dashpot constants are rescaled to the actual masses of the touching pair.)
159 // A wall acts as a particle of infinite mass, so the mass of a particle-wall
160 // contact is the particle mass itself rather than the reduced mass m/2 of a
161 // pair; that is why the wall constants below differ from the pair constants.
162 double mass_ref = p_parameters->density * (4.0 / 3.0) * PI * R_min * R_min * R_min; //mass_ref of a particle (later coefficients are corrected for real particle mass)
163 double tcontact = sqrt(0.5 * mass_ref * (PI * PI + pow(log(e_n_pp), 2)) / kn); //computed contact time
164 p_parameters->k_n_pp = kn; //normal elastic spring constant for particle-particle interactions
165 p_parameters->eta_n_pp = -2.0 * log(e_n_pp) * (0.5 * mass_ref) / tcontact; //normal dashpot damping constant for particle-particle interactions
166 p_parameters->k_t_pp = (mass_ref / 7.0) * (PI * PI + pow(log(e_t_pp), 2)) / (tcontact * tcontact); //tangential elastic spring constant for particle-particle interactions
167 p_parameters->eta_t_pp = -2.0 * log(e_t_pp) * (mass_ref / 7.0) / tcontact; //tangential dashpot damping coeff. for particle-particle interactions
168 p_parameters->fric_pp = muf; //friction coefficient for particle-particle interactions
169 for(unsigned int i=0; i< p_parameters->num_walls; ++i)
170 {
171 p_parameters->k_n_pw[i] = mass_ref * (PI * PI + pow(log(e_n_pw), 2)) / (tcontact * tcontact); //normal elastic spring constant for particle-wall interactions
172 p_parameters->eta_n_pw[i] = -2.0 * log(e_n_pw) * (mass_ref) / tcontact; //normal dashpot damping coeff. for particle-wall interactions
173 p_parameters->k_t_pw[i] = (2.0 * mass_ref / 7.0) * (PI * PI + pow(log(e_t_pw), 2)) / (tcontact * tcontact); //tangential elastic spring constant for particle-wall interactions
174 p_parameters->eta_t_pw[i] = -2.0 * log(e_t_pw) * (2.0 * mass_ref / 7.0) / tcontact; //tangential dashpot damping coeff. for particle-wall interactions
175 p_parameters->fric_pw[i] = muf_w;
176 }
177 p_parameters->e_n_pp_set = e_n_pp; // record what the contact parameters were tuned to
178 for (unsigned int i = 0; i < p_parameters->num_walls; ++i)
179 p_parameters->e_n_pw_set[i] = e_n_pw;
180 p_parameters->mass_ref = mass_ref; //mass_ref of a particle (later coefficients are corrected for real particle mass)
181
182 double v_small = 1e-2 * R_max/tcontact; //a velocity scale
183 p_parameters->Tg = 0.5 * mass_ref * v_small * v_small; //here Tg denotes the granular temperature, an average kinetic energy used for initialization
184 p_parameters->r_cut = 2.0 * R_max; //cut-off distance for pair interactions: the largest distance at which two particles can touch
185 p_parameters->dt = DT_FRACTION * tcontact; //integration time step: a contact must be resolved by many steps
186 p_parameters->r_shell = 0.2 * p_parameters->r_cut; //shell thickness for neighbor list
187 p_parameters->num_dt_steps = (size_t)(t_end / p_parameters->dt + 0.5);
188
189 if (p_parameters->r_cut > p_parameters->L.x / 2.0)
190 fprintf(stderr, "Warning! r_cut > Lx/2\n");
191 if (p_parameters->r_cut > p_parameters->L.y / 2.0)
192 fprintf(stderr, "Warning! r_cut > Ly/2\n");
193 if (p_parameters->r_cut > p_parameters->L.z / 2.0)
194 fprintf(stderr, "Warning! r_cut > Lz/2\n");
195}
