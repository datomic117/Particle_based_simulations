7// This function updates particle positions using their velocities.
8// The positions are advanced by one full time step (dt), and displacement vectors
9// (dr) for one time step are updated. The displacement since the last neighbor
10// list creation (stored in p_nbrlist->dr) is also updated for each particle.
11void update_positions(struct Parameters *p_parameters, struct Nbrlist *p_nbrlist, struct Vectors *p_vectors)
12{
13 struct Vec3D dr_loc;
14 struct Vec3D *r = p_vectors->r; // Particle positions
15 struct Vec3D *dr = p_vectors->dr; // Displacement in one timestep
16 struct Vec3D *v = p_vectors->v; // Particle velocities
17 struct DeltaR *dr_nbrlist = p_nbrlist->dr; // Displacement since last neighbor list creation
18 size_t num_part = p_parameters->num_part;
19 double dt = p_parameters->dt;
20
21 // Loop over all particles to update their positions
22 for (size_t i = 0; i < num_part; i++)
23 {
24 dr_loc.x = v[i].x * dt; // Compute displacement in x-direction for one timestep
25 dr_loc.y = v[i].y * dt; // Compute displacement in y-direction for one timestep
26 dr_loc.z = v[i].z * dt; // Compute displacement in z-direction for one timestep
27
28 dr[i] = dr_loc; // Store the displacement for this timestep
29 r[i].x += dr_loc.x; // Update position in x-direction
30 r[i].y += dr_loc.y; // Update position in y-direction
31 r[i].z += dr_loc.z; // Update position in z-direction
32
33 // Update the displacement since last neighbor list creation
34 dr_nbrlist[i].x += dr_loc.x;
35 dr_nbrlist[i].y += dr_loc.y;
36 dr_nbrlist[i].z += dr_loc.z;
37 dr_nbrlist[i].sq = (dr_nbrlist[i].x) * (dr_nbrlist[i].x) +
38 (dr_nbrlist[i].y) * (dr_nbrlist[i].y) +
39 (dr_nbrlist[i].z) * (dr_nbrlist[i].z); // Square of total displacement since last neighbor list creation
40 }
41}

43void update_tangential_displacements(struct Parameters *p_parameters, struct Vectors *p_vectors, struct Nbrlist *p_nbrlist, struct Colllist *p_colllist)
44{
45/* A sticking contact remembers how far the two surfaces have slid along each
46other: the accumulated tangential displacement is what the tangential spring
47of the contact model acts on. This function advances that memory by one time
48step for every current contact, particle-particle and particle-wall alike:
49the relative velocity of the two surfaces at the contact point, vijt (the
50tangential part of the velocity difference plus the contribution of the
51rotations), is integrated as tij(t+dt) = tij(t) + vijt(t+0.5*dt)*dt.
52The histories themselves live where the contact lives — beside the pair in
53the neighbor list (Nbrlist::tij) and in a fixed particle-wall slot
54(Colllist::tiw) — and are reset to zero by update_colllist when a contact
55ends. */
56 size_t i,j,k;
57 double fctr, dt = p_parameters->dt;
58 const size_t num_contacts = p_colllist->num_contacts;
59 const size_t *contact = p_colllist->contact;
60 struct DeltaR rij;
61 struct Vec3D vij, vijn, vijt;
62 struct Vec3D *v, *omega, *vw;
63 struct Pair * nbr;
64 struct DeltaR *tij;
65 v = p_vectors->v;
66 omega = p_vectors->omega;
67 nbr = p_nbrlist->nbr;
68 tij = p_nbrlist->tij;
69 vw = p_colllist->vw;
70
71 for (size_t c=0; c < num_contacts; c++)
72 {
73 // for each touching pair determine relative tangential velocity also taking into account rotation
74 k = contact[c];
75 rij = nbr[k].rij;
76 i = nbr[k].i;
77 j = nbr[k].j;
78 vij.x = v[i].x - v[j].x;
79 vij.y = v[i].y - v[j].y;
80 vij.z = v[i].z - v[j].z;
81 fctr = (vij.x*rij.x+vij.y*rij.y+vij.z*rij.z)/rij.sq;
82 vijn.x = fctr*rij.x;
83 vijn.y = fctr*rij.y;
84 vijn.z = fctr*rij.z;
85 vijt.x = vij.x-vijn.x;
86 vijt.y = vij.y-vijn.y;
87 vijt.z = vij.z-vijn.z;
88 vijt.x -= 0.5*((omega[i].y+omega[j].y)*rij.z-
89 (omega[i].z+omega[j].z)*rij.y);
90 vijt.y -= 0.5*((omega[i].z+omega[j].z)*rij.x-
91 (omega[i].x+omega[j].x)*rij.z);
92 vijt.z -= 0.5*((omega[i].x+omega[j].x)*rij.y-
93 (omega[i].y+omega[j].y)*rij.x);
94 // for each pair in the collisionlist update relative tangential displacements
95 tij[k].x += vijt.x*dt;
96 tij[k].y += vijt.y*dt;
97 tij[k].z += vijt.z*dt;
98 tij[k].sq = tij[k].x *tij[k].x + tij[k].y*tij[k].y + tij[k].z*tij[k].z;
99 }
100 size_t num_w = p_colllist->num_w;
101 size_t * indcs_w = p_colllist->indcs_w;
102 struct DeltaR * riw = p_colllist->riw;
103 struct DeltaR * tiw = p_colllist->tiw;
104 for (j=0; j < num_w; ++j)
105 {
106 i = indcs_w[j];
107 rij = riw[j];
108 vij.x = v[i].x-vw[j].x;
109 vij.y = v[i].y-vw[j].y;
110 vij.z = v[i].z-vw[j].z;
111 fctr = (vij.x*rij.x+vij.y*rij.y+vij.z*rij.z)/rij.sq;
112 vijn.x = fctr*rij.x;
113 vijn.y = fctr*rij.y;
114 vijn.z = fctr*rij.z;
115 vijt.x = vij.x-vijn.x;
116 vijt.y = vij.y-vijn.y;
117 vijt.z = vij.z-vijn.z;
118 vijt.x -= (omega[i].y*rij.z-omega[i].z*rij.y);
119 vijt.y -= (omega[i].z*rij.x-omega[i].x*rij.z);
120 vijt.z -= (omega[i].x*rij.y-omega[i].y*rij.x);
121 struct DeltaR *t = &tiw[i * NUM_WALLS_MAX + p_colllist->wall_id[j]];
122 t->x += vijt.x*dt;
123 t->y += vijt.y*dt;
124 t->z += vijt.z*dt;
125 t->sq = t->x * t->x + t->y * t->y + t->z * t->z;
126 }
127}

129// This function updates particle velocities by half a time step using the current forces.
130// The updated velocities are used in the velocity-Verlet integration scheme.
131// DEM particles also rotate: the angular velocities are advanced by the same
132// half-step using the torques, with I = (2/5) m R^2 the moment of inertia of a
133// solid sphere. The returned kinetic energy contains both the translational
134// and the rotational part.
135double update_velocities_half_dt(struct Parameters *p_parameters, struct Nbrlist *p_nbrlist, struct Vectors *p_vectors)
136{
137 (void)p_nbrlist; // unused here; the signature matches the other variants
138 double Ekin = 0.0, Ekin_rot = 0.0;
139 double *R = p_vectors->radius;
140 double *mass = p_vectors->mass;
141 const double factor = 0.5 * p_parameters->dt;
142 struct Vec3D *v, *omg, *f, *T;
143 v = p_vectors->v;
144 omg = p_vectors->omega;
145 f = p_vectors->f;
146 T = p_vectors->T;
147 for (size_t i = 0; i < p_parameters->num_part; i++)
148 {
149 double I = 0.4*mass[i]*R[i]*R[i];
150 v[i].x += factor * f[i].x/mass[i];
151 v[i].y += factor * f[i].y/mass[i];
152 v[i].z += factor * f[i].z/mass[i];
153 Ekin += mass[i]*(v[i].x * v[i].x + v[i].y * v[i].y + v[i].z * v[i].z);
154 omg[i].x += factor*T[i].x/I;
155 omg[i].y += factor*T[i].y/I;
156 omg[i].z += factor*T[i].z/I;
157 Ekin_rot += I*(omg[i].x*omg[i].x+omg[i].y*omg[i].y+omg[i].z*omg[i].z);
158 }
159 Ekin = 0.5* (Ekin + Ekin_rot);
160 return Ekin;
161}

163// This function applies periodic boundary conditions to ensure particles stay inside the simulation box.
164// If a particle moves beyond the box, it is wrapped around to the opposite side.
165void boundary_conditions(struct Parameters *p_parameters, struct Vectors *p_vectors)
166{
167 struct Vec3D invL; // Inverse of the box size
168 struct Vec3D *r = p_vectors->r; // Particle positions
169 struct Vec3D L = p_parameters->L; // Box dimensions
170 size_t num_part = p_parameters->num_part; // Number of particles
171
172 invL.x = 1.0 / L.x;
173 invL.y = 1.0 / L.y;
174 invL.z = 1.0 / L.z;
175
176 // Loop over all particles and apply periodic boundary conditions
177 for (size_t i = 0; i < num_part; i++)
178 {
179 r[i].x -= L.x * floor(r[i].x * invL.x); // Apply periodic boundary in x-direction
180 r[i].y -= L.y * floor(r[i].y * invL.y); // Apply periodic boundary in y-direction
181 r[i].z -= L.z * floor(r[i].z * invL.z); // Apply periodic boundary in z-direction
182 }
183}
