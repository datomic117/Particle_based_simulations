90static inline Vec3D v3_min_image(Vec3D r, Vec3D L)
91{
92 if (r.x >= 0.5 * L.x) r.x -= L.x; else if (r.x < -0.5 * L.x) r.x += L.x;
93 if (r.y >= 0.5 * L.y) r.y -= L.y; else if (r.y < -0.5 * L.y) r.y += L.y;
94 if (r.z >= 0.5 * L.z) r.z -= L.z; else if (r.z < -0.5 * L.z) r.z += L.z;
95 return r;
96}

102static inline Vec3D v3_in_box(Vec3D r, Vec3D L)
103{
104 r.x -= L.x * floor(r.x / L.x);
105 r.y -= L.y * floor(r.y / L.y);
106 r.z -= L.z * floor(r.z / L.z);
107 return r;
108}
