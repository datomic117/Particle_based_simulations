32// This function derives the complete molecular structure from the list of
33// bonds set up in initialise_bond_connectivity: every angle triplet i-j-k
34// (two bonds sharing atom j) and every dihedral quadruplet i-j-k-l (three
35// consecutive bonds), plus the 1-2, 1-3 and 1-4 partner lists that the
36// neighbor list uses to exclude or scale non-bonded interactions.
37//
38// The partner lists all use the same compact layout: the partners of particle
39// i are stored in pairs12[head12[i]] up to (excluding) pairs12[head12[i+1]].
40// Such a list is built in three passes: count the partners of each particle,
41// turn the counts into starting positions by a running sum, then fill the
42// slots. No searching or sorting is needed.
43void initialise_structure(struct Parameters *p_parameters, struct Vectors *p_vectors, struct Nbrlist *p_nbrlist)
44{
45 initialise_bond_connectivity(p_parameters, p_vectors); // Initialize bonds
46
47 struct Bond *bonds = p_vectors->bonds;
48 size_t num_bonds = p_vectors->num_bonds;
49 size_t num_part = p_parameters->num_part;
50
51 // Build the 1-2 partner list from the bonds. Each bond i-j contributes two
52 // entries: j is a partner of i and i is a partner of j.
53 size_t *cnt = (size_t *)calloc(num_part + 1, sizeof(size_t));
54 for (size_t i = 0; i < num_bonds; ++i)
55 {
56 ++cnt[bonds[i].i + 1];
57 ++cnt[bonds[i].j + 1];
58 }
59 size_t *head12 = (size_t *)malloc((num_part + 1) * sizeof(size_t));
60 head12[0] = 0;
61 for (size_t i = 1; i <= num_part; ++i)
62 {
63 head12[i] = cnt[i] + head12[i - 1]; // running sum: where the partners of particle i start
64 cnt[i] = head12[i]; // reused as the next free slot while filling
65 }
66 size_t *pairs12 = (size_t *)malloc(cnt[num_part] * sizeof(size_t));
67 for (size_t i = 0; i < num_bonds; ++i)
68 {
69 pairs12[cnt[bonds[i].i]++] = bonds[i].j;
70 pairs12[cnt[bonds[i].j]++] = bonds[i].i;
71 }
72
73 // An angle i-j-k is a pair of bonds sharing the central atom j, so each
74 // pair of 1-2 partners of an atom defines one angle centred on it.
75 size_t num_angles = 0;
76 for (size_t i = 1; i <= num_part; ++i)
77 {
78 size_t num_pairs = head12[i] - head12[i - 1];
79 num_angles += (num_pairs * (num_pairs - 1)) / 2;
80 }
81 struct Angle *angles = (struct Angle *)malloc(num_angles * sizeof(struct Angle));
82 size_t m = 0;
83 for (size_t i = 0; i < num_part; ++i)
84 for (size_t j = head12[i]; j < head12[i + 1]; ++j)
85 for (size_t k = j + 1; k < head12[i + 1]; ++k)
86 {
87 angles[m].i = pairs12[j];
88 angles[m].j = i;
89 angles[m].k = pairs12[k];
90 ++m;
91 }
92
93 // A dihedral i-j-k-l is built around a central bond j-k: i is any other
94 // partner of j, and l any other partner of k.
95 size_t num_dihdr = 0;
96 for (size_t i = 0; i < num_bonds; ++i)
97 {
98 num_dihdr += (head12[bonds[i].i + 1] - head12[bonds[i].i] - 1) * (head12[bonds[i].j + 1] - head12[bonds[i].j] - 1);
99 }
100 struct Dihedral *dihedrals = (struct Dihedral *) malloc(num_dihdr * sizeof(struct Dihedral));
101 size_t k = 0;
102 for (size_t i = 0; i < num_bonds; ++i)
103 {
104 struct Dihedral dihdr;
105 dihdr.j = bonds[i].i;
106 dihdr.k = bonds[i].j;
107 for (size_t p = head12[dihdr.j]; p < head12[dihdr.j + 1]; ++p)
108 {
109 if (pairs12[p] == dihdr.k)
110 continue;
111 dihdr.i = pairs12[p];
112 for (size_t q = head12[dihdr.k]; q < head12[dihdr.k + 1]; ++q)
113 {
114 if (pairs12[q] != dihdr.j)
115 {
116 dihdr.l = pairs12[q];
117 dihedrals[k++] = dihdr;
118 }
119 }
120 }
121 }
122 // 1-3 partners are the two outer atoms of each angle. The list is only
123 // needed when the 1-3 non-bonded interaction is scaled (factor != 1).
124 if (p_parameters->factor_13_nb != 1.0)
125 {
126 for (size_t i = 0; i <= num_part; ++i)
127 cnt[i] = 0;
128 for (size_t i = 0; i < num_angles; ++i)
129 {
130 ++cnt[angles[i].i + 1];
131 ++cnt[angles[i].k + 1];
132 }
133 size_t *head13 = (size_t *)malloc((num_part + 1) * sizeof(size_t));
134 head13[0] = 0;
135 for (size_t i = 1; i <= num_part; ++i)
136 {
137 head13[i] = cnt[i] + head13[i - 1];
138 cnt[i] = head13[i];
139 }
140 size_t *pairs13 = (size_t *)malloc(cnt[num_part] * sizeof(size_t));
141 for (size_t i = 0; i < num_angles; ++i)
142 {
143 pairs13[cnt[angles[i].i]++] = angles[i].k;
144 pairs13[cnt[angles[i].k]++] = angles[i].i;
145 }
146 p_nbrlist->head13 = head13;
147 p_nbrlist->pairs13 = pairs13;
148 }
149 // 1-4 partners are the two outer atoms of each dihedral, built the same way.
150 if (p_parameters->factor_14_nb != 1.0)
151 {
152 for (size_t i = 0; i <= num_part; ++i)
153 cnt[i] = 0;
154 for (size_t i = 0; i < num_dihdr; ++i)
155 {
156 ++cnt[dihedrals[i].i + 1];
157 ++cnt[dihedrals[i].l + 1];
158 }
159 size_t *head14 = (size_t *)malloc((num_part + 1) * sizeof(size_t));
160 head14[0] = 0;
161 for (size_t i = 1; i <= num_part; ++i)
162 {
163 head14[i] = cnt[i] + head14[i - 1];
164 cnt[i] = head14[i];
165 }
166 size_t *pairs14 = (size_t *)malloc(cnt[num_part] * sizeof(size_t));
167 for (size_t i = 0; i < num_dihdr; ++i)
168 {
169 pairs14[cnt[dihedrals[i].i]++] = dihedrals[i].l;
170 pairs14[cnt[dihedrals[i].l]++] = dihedrals[i].i;
171 }
172 p_nbrlist->head14 = head14;
173 p_nbrlist->pairs14 = pairs14;
174 }
175
176 p_vectors->num_angles = num_angles;
177 p_vectors->angles = angles;
178 p_vectors->num_dihedrals = num_dihdr;
179 p_vectors->dihedrals = dihedrals;
180 // The 1-2 partner list is only kept if the neighbor list needs it to scale
181 // the non-bonded interaction of bonded pairs.
182 if (p_parameters->factor_12_nb != 1.0)
183 {
184 p_nbrlist->head12 = head12;
185 p_nbrlist->pairs12 = pairs12;
186 }
187 else
188 {
189 free(head12);
190 free(pairs12);
191 }
192 free(cnt);
193}

211// This function initializes particle positions on a cubic lattice.
212// Particles are placed in a grid with spacing based on the number of particles and the box dimensions.
213void initialise_positions(struct Parameters *p_parameters, struct Vectors *p_vectors)
214{
215 struct Vec3D dr; // Displacement vector for positioning particles
216 struct Index3D n; // Number of grid cells along each axis
217 double dl; // Lattice spacing
218 size_t ipart = 0; // Particle index; size_t, like the count it is compared
219 // against and the arrays it indexes
220
223
224 // Calculate lattice spacing based on particle number and box dimensions
225 dl = pow(p_parameters->L.x * p_parameters->L.y * p_parameters->L.z / ((double)p_parameters->num_part), 1.0 / 3.0);
226 n.i = (int)ceil(p_parameters->L.x / dl);
227 n.j = (int)ceil(p_parameters->L.y / dl);
228 n.k = (int)ceil(p_parameters->L.z / dl);
229 dr.x = p_parameters->L.x / (double)n.i;
230 dr.y = p_parameters->L.y / (double)n.j;
231 dr.z = p_parameters->L.z / (double)n.k;
232 ipart = 0;
233 for (size_t i = 0; i < n.i; ++i)
234 for (size_t j = 0; j < n.j; ++j)
235 for (size_t k = 0; k < n.k; ++k, ++ipart)
236 {
237 if (ipart >= p_parameters->num_part)
238 break;
239 p_vectors->r[ipart].x = (i + 0.5) * dr.x;
240 p_vectors->r[ipart].y = (j + 0.5) * dr.y;
241 p_vectors->r[ipart].z = (k + 0.5) * dr.z;
242 }
243}

245// This function initializes the velocities of particles based on the Maxwell-Boltzmann distribution.
246// The total momentum is also removed to ensure zero total momentum (important for stability).
247void initialise_velocities(struct Parameters *p_parameters, struct Vectors *p_vectors)
248{
251 double sqrtktm = sqrt(p_parameters->kT / p_parameters->mass);
252 struct Vec3D sumv = {0.0, 0.0, 0.0}; // Total velocity (to remove later)
253 Vec3D *v = p_vectors->v; // Pointer to particle velocities
254
255 // Assign random velocities to each particle
256 for (size_t i = 0; i < p_parameters->num_part; i++)
257 {
258 v[i] = v3_scl(sqrtktm, v3(gauss(), gauss(), gauss()));
259 sumv = v3_add(sumv, v[i]);
260 }
261
262 // Remove the average velocity to ensure zero total momentum
263 sumv = v3_scl(1.0 / (double)p_parameters->num_part, sumv);
264 for (size_t i = 0; i < p_parameters->num_part; i++)
265 {
266 v[i] = v3_sub(v[i], sumv);
267 }
268}
