67 struct Vectors vectors;
68 struct Parameters parameters;
69 struct Nbrlist nbrlist;
70 size_t step;
71 double Ekin, Epot, time;
72
73 // Step 1: Set the simulation parameters from input files
74 set_parameters(&parameters);
75
76 // Step 2: Allocate memory for particles, forces, and neighbor lists
77 alloc_memory(&parameters, &vectors, &nbrlist);
78
79 // Check if a restart is required. A restart file stores only positions,
80 // velocities and forces, so the types and the molecular topology are
81 // reconstructed here just as in a fresh initialisation.
82 if (parameters.load_restart == 1)
83 {
84 load_restart(&parameters, &vectors);
85 boundary_conditions(&parameters, &vectors);
86 initialise_types(&parameters, &vectors);
87 initialise_structure(&parameters, &vectors, &nbrlist);
88 step = 0;
89 time = 0.0;
90 }
91 else
92 {
95 initialise(&parameters, &vectors, &nbrlist, &step, &time);
96 boundary_conditions(&parameters, &vectors);
97 }
98
99 // Step 3: Build the neighbor list for non-bonded interactions
100 build_nbrlist(&parameters, &vectors, &nbrlist);
101
102 // Step 4: Calculate initial forces (non-bonded and bonded if implemented)
103 Epot = calculate_forces(&parameters, &nbrlist, &vectors);
104
105 // Output initial particle positions in PDB format
106 record_trajectories_pdb(1, &parameters, &vectors, time);
107
108 // In force-test mode the forces and the virial are verified against finite
109 // differences of the potential energy, after which the program exits. Every
110 // force_test-th particle is tested, so force_test = 1 tests all of them.
111 // Use this to check every force term you implement: the analytical force
112 // must equal minus the numerical derivative of the potential energy.
113 //
114 // forces_test takes the routine to check as its first argument, so a term
115 // can be tested on its own instead of inside the sum of all of them. That
116 // is what task B5 asks for, and it matters: an error in one term can hide
117 // behind a larger correct one in the total. Write a wrapper per term with
118 // the signature of calculate_forces, for example
119 //
120 // double forces_bond_only(struct Parameters *p_parameters,
121 // struct Nbrlist *p_nbrlist,
122 // struct Vectors *p_vectors)
123 // {
124 // (void)p_nbrlist; // this term needs no pairs
125 // for (size_t i = 0; i < p_parameters->num_part; i++)
126 // p_vectors->f[i] = (struct Vec3D){0.0, 0.0, 0.0};
127 // return calculate_forces_bond(p_parameters, p_vectors);
128 // }
129 //
130 // and pass it here in place of calculate_forces.
131 if (parameters.force_test > 0)
132 {
133 for(size_t i=0; i<parameters.num_part; i+=(size_t)parameters.force_test)
134 forces_test(calculate_forces, (int)i, &parameters, &nbrlist, &vectors);
135 virial_test(calculate_forces, &parameters, &nbrlist, &vectors);
136 free_memory(&vectors, &nbrlist);
137 return 0;
138 }
139
140 // Main MD loop using velocity-Verlet integration
141 while (step < parameters.num_dt_steps)
142 {
143 step++;
144 time += parameters.dt;
145
146 // Update velocities (half-step)
148 Ekin = update_velocities_half_dt(&parameters, &nbrlist, &vectors);
149
151 if (parameters.is_NVT == 1)
152 thermostat(&parameters, &vectors, Ekin);
153
154 // Update positions
155 update_positions(&parameters, &nbrlist, &vectors);
156
157 // Apply boundary conditions
158 boundary_conditions(&parameters, &vectors);
159
160 // Rebuild neighbor list if needed
161 update_nbrlist(&parameters, &vectors, &nbrlist);
162
163 // Calculate forces for the current configuration (bonded forces if implemented)
164 Epot = calculate_forces(&parameters, &nbrlist, &vectors);
165
166 // Final velocity update (half-step)
167 Ekin = update_velocities_half_dt(&parameters, &nbrlist, &vectors);
168
169 // Output system state every 'num_dt_pdb' steps
170 if (step % parameters.num_dt_pdb == 0)
171 record_trajectories_pdb(0, &parameters, &vectors, time);
172
173 // Save restart file every 'num_dt_restart' steps
174 if (step % parameters.num_dt_restart == 0)
175 save_restart(&parameters, &vectors);
176
178 // Print to the screen to monitor the progress of the simulation
179 if (step % parameters.num_dt_output == 0)
180 {
182 printf("Step %lu, Time %f, Epot %f, Ekin %f, Etot %f\n", (long unsigned)step, time, Epot, Ekin, Epot + Ekin);
183 }
184 }
185
186 // Save final state
187 save_restart(&parameters, &vectors);
188
189 // Step 5: Free memory and clean up
190 free_memory(&vectors, &nbrlist);
191
192 return 0;
193}
