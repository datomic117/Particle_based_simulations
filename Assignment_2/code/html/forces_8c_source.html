9// This function calculates all forces acting on the particles (bonded and non-bonded).
10// It initializes the forces array, then calculates bond-stretch, angle-bend, dihedral-torsion,
11// and non-bonded forces. The total potential energy is returned.
12double calculate_forces(struct Parameters *p_parameters, struct Nbrlist *p_nbrlist, struct Vectors *p_vectors)
13{
14 struct Vec3D *f = p_vectors->f;
15 size_t num_part = p_parameters->num_part;
16
17 // Initialize the forces to zero for all particles
18 for (size_t i = 0; i < num_part; i++)
19 f[i] = (struct Vec3D){0.0, 0.0, 0.0};
20
21 // Initialize the virial accumulators; each force routine adds its f.r contributions
22 p_vectors->press_vir_bnd = 0.0;
23 p_vectors->press_vir_nb = 0.0;
24
25 // Calculate the forces and accumulate the potential energy from each type of interaction
26 double Epot = 0.0;
27 Epot += calculate_forces_bond(p_parameters, p_vectors);
28 Epot += calculate_forces_angle(p_parameters, p_vectors);
29 Epot += calculate_forces_dihedral(p_parameters, p_vectors);
30 Epot += calculate_forces_nb(p_parameters, p_nbrlist, p_vectors);
31
32 // Normalize the accumulated virials W = sum(f.r) to pressure contributions W/(3V)
33 double V = p_parameters->L.x * p_parameters->L.y * p_parameters->L.z;
34 p_vectors->press_vir_bnd /= (3.0 * V);
35 p_vectors->press_vir_nb /= (3.0 * V);
36
37 return Epot;
38}

40// This function calculates bond-stretch forces based on the current positions of the bonded particles.
41// It applies the minimum image convention to calculate the distance between bonded pairs and then
42// computes the force and potential energy due to the bond interaction.
43double calculate_forces_bond(struct Parameters *p_parameters, struct Vectors *p_vectors)
44{
45 double Epot = 0.0;
46 struct Bond *bonds = p_vectors->bonds;
47 size_t num_bonds = p_vectors->num_bonds;
48 struct Vec3D *f = p_vectors->f;
49 struct Vec3D *r = p_vectors->r;
50 struct Vec3D L = p_parameters->L;
51 struct Vec3D rij;
52 struct Vec3D fi = {0.0, 0.0, 0.0};
53
54 // Loop through each bond and calculate the forces
55 for (size_t q = 0; q < num_bonds; ++q)
56 {
57 size_t i = bonds[q].i;
58 size_t j = bonds[q].j;
59
60 // Apply the minimum image convention for calculating distances
61 rij = v3_min_image(v3_sub(r[i], r[j]), L);
62
64 // Newton's third law: particle j receives the opposite force
65 f[i] = v3_add(f[i], fi);
66 f[j] = v3_sub(f[j], fi);
67 }
68
69 return Epot; // Return the potential energy due to bond-stretch interactions
70}

72// This function calculates angle-bend forces based on the current positions of the angle-defined particles.
73// It uses the minimum image convention and computes forces due to angle interactions.
74double calculate_forces_angle(struct Parameters *p_parameters, struct Vectors *p_vectors)
75{
76 double Epot = 0.0;
77 struct Angle *angles = p_vectors->angles;
78 size_t num_angles = p_vectors->num_angles;
79 struct Vec3D *f = p_vectors->f;
80 struct Vec3D *r = p_vectors->r;
81 struct Vec3D L = p_parameters->L;
82 struct Vec3D rij, rkj;
83 struct Vec3D fi = {0.0, 0.0, 0.0}, fk = {0.0, 0.0, 0.0};
84
85 // Loop through each angle and calculate the forces
86 // Note: The angles triplets ijk are computed from the bond information during initialization. This is already implemented.
87 for (size_t q = 0; q < num_angles; ++q)
88 {
89 size_t i = angles[q].i;
90 size_t j = angles[q].j;
91 size_t k = angles[q].k;
92
93 // Apply the minimum image convention for calculating distances
94 rij = v3_sub(r[i], r[j]);
95 rij = v3_min_image(rij, L);
96 rkj = v3_sub(r[k], r[j]);
97 rkj = v3_min_image(rkj, L);
98
100
101 // The central atom takes the opposite of both outer forces, so the
102 // total force of the angle interaction is zero
103 f[i] = v3_add(f[i], fi);
104 f[j] = v3_sub(f[j], fi);
105 f[j] = v3_sub(f[j], fk);
106 f[k] = v3_add(f[k], fk);
107 }
108 return Epot; // Return the potential energy due to angle-bend interactions
109}

123
124 // Loop through each dihedral and calculate the forces
125 // Note: The dihedrals are computed from the bond information during initialization. This is already implemented.
126 for (size_t q = 0; q < num_dihedrals; ++q)
127 {
128 size_t i = dihedrals[q].i;

131 size_t l = dihedrals[q].l;
132

139 rkj = v3_min_image(rkj, L);
140 rkl = v3_sub(r[k], r[l]);
141 rkl = v3_min_image(rkl, L);

144 }

149// This function calculates non-bonded forces between particles using the neighbor list.
150// The potential energy and forces are calculated using the Lennard-Jones potential.
151double calculate_forces_nb(struct Parameters *p_parameters, struct Nbrlist *p_nbrlist, struct Vectors *p_vectors)
152{
153 struct Vec3D df;
154 double r_cutsq, sr2, sr6, sr12, fr;
155 struct DeltaR rij;
156 struct Pair *nbr = p_nbrlist->nbr;
157 const size_t num_nbrs = p_nbrlist->num_nbrs;
158 struct Vec3D *f = p_vectors->f;
159 struct Vec3D *r_pos = p_vectors->r;
160 const struct Vec3D L = p_parameters->L;
161
162 r_cutsq = p_parameters->r_cut * p_parameters->r_cut;
163 double Epot = 0.0;
164
165 double sigmasq = p_parameters->sigma * p_parameters->sigma;
166 double epsilon = p_parameters->epsilon;
167
168 double Epot_cutoff;
169 sr2 = sigmasq / r_cutsq;
170 sr6 = sr2 * sr2 * sr2;
171 sr12 = sr6 * sr6;
172 Epot_cutoff = sr12 - sr6;
173
174 // Loop through the neighbor list and calculate the forces for each particle pair
175 for (size_t k = 0; k < num_nbrs; k++)
176 {
177 size_t i = nbr[k].i;
178 size_t j = nbr[k].j;
179 // The pair list holds only the pairs; the connecting vector is taken
180 // from the current positions. A pair is far closer than half a box, so
181 // the cheap minimum image applies.
182 rij.v = v3_min_image(v3_sub(r_pos[i], r_pos[j]), L);
183 rij.sq = v3_dot(rij.v, rij.v);
184 // Compute forces if the distance is smaller than the cutoff distance
185 if (rij.sq < r_cutsq)
186 {
187 double factor = nbr[k].factor; // 0 or 1, or the 1-4 scaling factor
189
190 // The LJ powers are computed by squaring: (sigma/r)^6 and
191 // (sigma/r)^12 from (sigma/r)^2, with no call to pow()
192 sr2 = sigmasq / rij.sq;
193 sr6 = sr2 * sr2 * sr2;
194 sr12 = sr6 * sr6;
195
196 // Calculate the potential energy, shifted to zero at the cutoff
197 Epot += factor * 4.0 * epsilon * (sr12 - sr6 - Epot_cutoff);
198
199 // Compute the force and apply it to both particles
200 fr = factor * 24.0 * epsilon * (2.0 * sr12 - sr6) / rij.sq; // Force divided by distance: multiplying by the vector rij then gives the force, with no sqrt needed
201 p_vectors->press_vir_nb += fr * rij.sq; // virial contribution f_ij . r_ij
202 df = v3_scl(fr, rij.v);
203 // Update forces on particles i and j
204 f[i] = v3_add(f[i], df);
205 f[j] = v3_sub(f[j], df);
206 }
207 }
208 return Epot; // Return the potential energy due to non-bonded interactions
209}
